\begin{description}[style=nextline]
  \item[AI] Artificial Intelligence / sztuczna inteligencja
  \item[ASP] Akademia Sztuki w Szczecinie
  \item[Big Five] Model Wielkiej Piątki cech osobowości
  \item[ComfyUI] Środowisko grafów do eksperymentów z modelami generatywnymi (pracownia autora; nie runtime IDA)
  \item[ControlNet] Conditioning struktury obrazu (depth, krawędzie itd.) --- warstwa pracowni
  \item[FLUX] Rodzina modeli dyfuzyjnych używana we wczesnej fazie (Modal) --- genealogia, nie stan obecny produkcji
  \item[GCP] Google Cloud Platform
  \item[HCI] Human--Computer Interaction
  \item[IAT] Implicit Association Test
  \item[IDA] AI Interior Design Dialogue Research Platform
  \item[IPIP-NEO-120] International Personality Item Pool (120 pozycji, 30 facetów)
  \item[LLM] Large Language Model
  \item[PRS] Perceived Restorativeness Scale
  \item[RQ] Research Question / pytanie badawcze (w pracy: RQ1--RQ6)
  \item[RtD] Research through Design / badania przez projektowanie
  \item[SAM / SAM2] Segment Anything Model --- segmentacja w pracowni generatywnej
  \item[Vertex AI] Platforma Google Cloud do modeli generatywnych; produkcyjna generacja obrazu IDA (Gemini Image)
  \item[VLM] Vision--Language Model
  \item[XAI] Explainable Artificial Intelligence
\end{description}
